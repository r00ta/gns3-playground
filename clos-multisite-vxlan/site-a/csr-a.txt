hostname CSR-A

! WAN interface
interface GigabitEthernet1
 description WAN
 ip address 100.0.0.1 255.255.255.252
 no shutdown

! Fabric-facing interface
interface GigabitEthernet2
 description toFabric-SiteA
 ip address 10.99.0.1 255.255.255.0
 no shutdown

! Loopback
interface Loopback0
 ip address 10.0.0.100 255.255.255.255

! IPsec SVTI tunnel to Site B
interface Tunnel100
 description IPsec-to-CSR-B
 ip address 172.16.100.1 255.255.255.252
 ip mtu 1400
 ip tcp adjust-mss 1360
 tunnel source GigabitEthernet1
 tunnel destination 100.0.0.2
 tunnel mode ipsec ipv4
 tunnel protection ipsec profile SITE-PROFILE
 no shutdown

! IKEv2
crypto ikev2 proposal IKEV2-PROP
 encryption aes-cbc-256
 integrity sha256
 group 14

crypto ikev2 policy IKEV2-POL
 proposal IKEV2-PROP

crypto ikev2 keyring KEYRING
 peer CSR-B
  address 100.0.0.2
  pre-shared-key MyStr0ngSecret!

crypto ikev2 profile IKEV2-PROF
 match identity remote address 100.0.0.2 255.255.255.255
 authentication remote pre-share
 authentication local pre-share
 keyring local KEYRING

crypto ipsec transform-set TS esp-aes 256 esp-sha256-hmac
 mode tunnel

crypto ipsec profile SITE-PROFILE
 set transform-set TS
 set ikev2-profile IKEV2-PROF

! OSPF underlay — advertise loopback, fabric link, and tunnel link
router ospf 1
 router-id 10.0.0.100
 network 10.0.0.100 0.0.0.0 area 0
 network 10.99.0.0 0.0.0.255 area 0
 network 172.16.100.0 0.0.0.3 area 0
